# Speichern und Laden in der Partie unter Wine (Testdaten), ueber das Lade-Fenster des Spielmenues.
# Aufruf: tools/wine/run_game.sh <Datenordner> -skirmish TestSkirmish -skirmishAI 2 -testScript laden_im_spiel.txt -crcLog x.crc
# (die Datei liegt dazu im Datenordner). Eintrag 0 der Liste ist "Neuer Spielstand", 1 der juengste Spielstand.
# Bleibt der Ablauf bei "klick PopupSaveLoad.wnd:ButtonLoad" stehen, ist die Liste leer (Spielstaende nicht gefunden).
game
wait 60
save Wine_Testablauf.sav
wait 30
gameMenu
click QuitMenu.wnd:ButtonSaveLoad
window PopupSaveLoad.wnd:ListboxGames
select PopupSaveLoad.wnd:ListboxGames 1
wait 10
click PopupSaveLoad.wnd:ButtonLoad
clickIf PopupSaveLoad.wnd:ButtonLoadConfirm
game
wait 60
quit
