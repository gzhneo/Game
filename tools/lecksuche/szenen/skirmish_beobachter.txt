# Skirmish als Beobachter: Platz 0 auf "Beobachter" (letzter Eintrag der Seitenauswahl, -1),
# Platz 1 klassische KI (schwer), Platz 2 neue KI (Brutal). Partie 1 mit Speichern und Laden als Beobachter,
# Spielende per Sieg, Punkteanzeige, zurueck ins Skirmish-Menue; Partie 2 ebenso. Danach Platz 0 wieder auf
# Zufall und Platz 2 geschlossen (das Menue behaelt die Wahl in den Skirmish-Einstellungen), Hauptmenue, beenden.
# Karte: die zuletzt im Skirmish-Menue gewaehlte (der Beobachter braucht keinen Startplatz, zwei genuegen).
# Der Spielstand heisst Lecksuche_Beobachter.sav; tools/lecksuche loescht ihn.
mainMenu
clickIf MainMenu.wnd:ButtonSinglePlayer
click MainMenu.wnd:ButtonSkirmish
window SkirmishGameOptionsMenu.wnd:ButtonStart
select SkirmishGameOptionsMenu.wnd:ComboBoxPlayerTemplate0 -1
select SkirmishGameOptionsMenu.wnd:ComboBoxPlayer1 4
select SkirmishGameOptionsMenu.wnd:ComboBoxPlayer2 5
wait 10
# Partie 1 mit Speichern und Laden
click SkirmishGameOptionsMenu.wnd:ButtonStart
game
wait 1800
save Lecksuche_Beobachter.sav
wait 300
load Lecksuche_Beobachter.sav
game
wait 900
win
noGame
click ScoreScreen.wnd:ButtonOk
window SkirmishGameOptionsMenu.wnd:ButtonStart
wait 60
# Partie 2
click SkirmishGameOptionsMenu.wnd:ButtonStart
game
wait 1800
win
noGame
click ScoreScreen.wnd:ButtonOk
window SkirmishGameOptionsMenu.wnd:ButtonStart
wait 60
select SkirmishGameOptionsMenu.wnd:ComboBoxPlayerTemplate0 0
select SkirmishGameOptionsMenu.wnd:ComboBoxPlayer2 1
wait 10
click SkirmishGameOptionsMenu.wnd:ButtonBack
mainMenu
wait 60
quit
