# Menues durchklicken, jedes Untermenue auf und wieder zu. Ende ueber den Knopf "Beenden" des
# Hauptmenues (quitCallback), also den Weg, den ein Spieler nimmt.
mainMenu
# Optionen (die Tastaturbelegung ist in den EA-Daten verborgen)
click MainMenu.wnd:ButtonOptions
window OptionsMenu.wnd:ButtonBack
wait 60
click OptionsMenu.wnd:ButtonBack
mainMenu
# Mehrspieler: LAN-Lobby, Direktverbindung, eigenes LAN-Spiel eroeffnen und wieder verlassen
clickIf MainMenu.wnd:ButtonMultiplayer
click MainMenu.wnd:ButtonNetwork
window LanLobbyMenu.wnd:ButtonBack
wait 300
click LanLobbyMenu.wnd:ButtonDirectConnect
window NetworkDirectConnect.wnd:ButtonBack
wait 30
click NetworkDirectConnect.wnd:ButtonBack
window LanLobbyMenu.wnd:ButtonHost
wait 30
click LanLobbyMenu.wnd:ButtonHost
window LanGameOptionsMenu.wnd:ButtonBack
wait 300
click LanGameOptionsMenu.wnd:ButtonBack
window LanLobbyMenu.wnd:ButtonBack
wait 30
click LanLobbyMenu.wnd:ButtonBack
mainMenu
clickIf MainMenu.wnd:ButtonMultiBack
mainMenu
# Replays und Spielstaende
clickIf MainMenu.wnd:ButtonLoadReplay
click MainMenu.wnd:ButtonReplay
window ReplayMenu.wnd:ButtonBack
wait 60
click ReplayMenu.wnd:ButtonBack
mainMenu
clickIf MainMenu.wnd:ButtonLoadReplay
click MainMenu.wnd:ButtonLoadGame
window SaveLoad.wnd:ButtonBack
wait 60
click SaveLoad.wnd:ButtonBack
mainMenu
clickIf MainMenu.wnd:ButtonLoadReplayBack
mainMenu
# Einzelspieler: Skirmish-Menue auf und zu
clickIf MainMenu.wnd:ButtonSinglePlayer
click MainMenu.wnd:ButtonSkirmish
window SkirmishGameOptionsMenu.wnd:ButtonBack
wait 60
click SkirmishGameOptionsMenu.wnd:ButtonBack
mainMenu
clickIf MainMenu.wnd:ButtonSingleBack
mainMenu
# Mitwirkende: eine Weile laufen lassen, dann Escape
clickIf MainMenu.wnd:ButtonSingleBack
clickIf MainMenu.wnd:ButtonLoadReplayBack
clickIf MainMenu.wnd:ButtonMultiBack
mainMenu
click MainMenu.wnd:ButtonCredits
window CreditsMenu.wnd:ParentCreditsWindow
wait 900
esc CreditsMenu.wnd:ParentCreditsWindow
mainMenu
wait 60
clickIf MainMenu.wnd:ButtonSingleBack
clickIf MainMenu.wnd:ButtonLoadReplayBack
clickIf MainMenu.wnd:ButtonMultiBack
mainMenu
click MainMenu.wnd:ButtonExit
wait 600
log Beenden-Knopf ohne Wirkung
quit
