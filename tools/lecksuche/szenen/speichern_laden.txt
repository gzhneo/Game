# Speichern und Laden: Skirmish (USA gegen eine KI), speichern, weiterspielen, in der Partie laden,
# weiterspielen, Partie ueber Strg+W beenden, Hauptmenue; dann aus dem Hauptmenue ueber das Lade-Menue
# laden (erster Eintrag = juengster Spielstand), weiterspielen, beenden.
# Der Spielstand heisst Lecksuche_Testablauf.sav (Spielstand-Ordner); tools/lecksuche loescht ihn danach.
mainMenu
clickIf MainMenu.wnd:ButtonSinglePlayer
click MainMenu.wnd:ButtonSkirmish
window SkirmishGameOptionsMenu.wnd:ButtonStart
select SkirmishGameOptionsMenu.wnd:ComboBoxPlayerTemplate0 1
select SkirmishGameOptionsMenu.wnd:ComboBoxPlayer1 2
wait 10
click SkirmishGameOptionsMenu.wnd:ButtonStart
game
wait 1800
save Lecksuche_Testablauf.sav
wait 600
load Lecksuche_Testablauf.sav
game
wait 600
win
noGame
click ScoreScreen.wnd:ButtonOk
click SkirmishGameOptionsMenu.wnd:ButtonBack
mainMenu
wait 60
# Laden aus dem Hauptmenue ueber das Lade-Menue
clickIf MainMenu.wnd:ButtonSingleBack
mainMenu
clickIf MainMenu.wnd:ButtonLoadReplay
click MainMenu.wnd:ButtonLoadGame
window SaveLoad.wnd:ListboxGames
select SaveLoad.wnd:ListboxGames 0
wait 10
click SaveLoad.wnd:ButtonLoad
game
wait 600
quit
