# Im Fenster mit Zeichnen: Optionsmenue, Detailstufe Niedrig (Eintrag 3) uebernehmen -- die
# D3D11-Nachbearbeitung schaltet ab --, Skirmish-Partie, dann Ultra (Eintrag 0) uebernehmen, Partie, Hoch
# (Eintrag 1); dazu ein Aufloesungswechsel hin (Eintrag 1) und zurueck (Eintrag 0) mit Bestaetigung. Die
# Einstellungen des Nutzers stellt tools/lecksuche nach dem Lauf wieder her.
mainMenu
click MainMenu.wnd:ButtonOptions
window OptionsMenu.wnd:ButtonAccept
list OptionsMenu.wnd:ComboBoxResolution
select OptionsMenu.wnd:ComboBoxDetail 3
wait 10
click OptionsMenu.wnd:ButtonAccept
mainMenu
wait 300
# Partie auf Niedrig
clickIf MainMenu.wnd:ButtonSinglePlayer
click MainMenu.wnd:ButtonSkirmish
window SkirmishGameOptionsMenu.wnd:ButtonStart
select SkirmishGameOptionsMenu.wnd:ComboBoxPlayerTemplate0 1
select SkirmishGameOptionsMenu.wnd:ComboBoxPlayer1 2
wait 10
click SkirmishGameOptionsMenu.wnd:ButtonStart
game
wait 600
win
noGame
click ScoreScreen.wnd:ButtonOk
click SkirmishGameOptionsMenu.wnd:ButtonBack
mainMenu
# Ultra
clickIf MainMenu.wnd:ButtonSingleBack
mainMenu
click MainMenu.wnd:ButtonOptions
window OptionsMenu.wnd:ButtonAccept
select OptionsMenu.wnd:ComboBoxDetail 0
wait 10
click OptionsMenu.wnd:ButtonAccept
mainMenu
wait 300
clickIf MainMenu.wnd:ButtonSinglePlayer
click MainMenu.wnd:ButtonSkirmish
window SkirmishGameOptionsMenu.wnd:ButtonStart
wait 10
click SkirmishGameOptionsMenu.wnd:ButtonStart
game
wait 600
win
noGame
click ScoreScreen.wnd:ButtonOk
click SkirmishGameOptionsMenu.wnd:ButtonBack
mainMenu
# Aufloesung wechseln und bestaetigen, dann zurueck
clickIf MainMenu.wnd:ButtonSingleBack
mainMenu
click MainMenu.wnd:ButtonOptions
window OptionsMenu.wnd:ButtonAccept
select OptionsMenu.wnd:ComboBoxResolution 1
wait 10
click OptionsMenu.wnd:ButtonAccept
clickIf MessageBox.wnd:ButtonOk 600
mainMenu
wait 300
click MainMenu.wnd:ButtonOptions
window OptionsMenu.wnd:ButtonAccept
select OptionsMenu.wnd:ComboBoxResolution 0
select OptionsMenu.wnd:ComboBoxDetail 1
wait 10
click OptionsMenu.wnd:ButtonAccept
clickIf MessageBox.wnd:ButtonOk 600
mainMenu
wait 300
quit
