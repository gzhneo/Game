# Skirmish (China gegen eine leichte KI): im Spielmenue die Optionen und das Speichern/Laden-Fenster oeffnen
# und schliessen, die Partie neu starten ("Neustart", Rueckfrage mit Ja), weiterspielen, dann ueber das
# Spielmenue verlassen, Punkteanzeige, Hauptmenue.
mainMenu
clickIf MainMenu.wnd:ButtonSinglePlayer
click MainMenu.wnd:ButtonSkirmish
window SkirmishGameOptionsMenu.wnd:ButtonStart
select SkirmishGameOptionsMenu.wnd:ComboBoxPlayerTemplate0 2
select SkirmishGameOptionsMenu.wnd:ComboBoxPlayer1 2
wait 10
click SkirmishGameOptionsMenu.wnd:ButtonStart
game
wait 900
gameMenu
click QuitMenu.wnd:ButtonOptions
window OptionsMenu.wnd:ButtonBack
wait 30
click OptionsMenu.wnd:ButtonBack
wait 30
clickIf QuitMenu.wnd:ButtonSaveLoad
window PopupSaveLoad.wnd:ButtonBack 300
wait 30
click PopupSaveLoad.wnd:ButtonBack
wait 30
click QuitMenu.wnd:ButtonRestart
click MessageBox.wnd:ButtonYes
wait 60
game
wait 900
gameMenu
click QuitMenu.wnd:ButtonExit
click QuitMessageBox.wnd:ButtonYes
noGame
click ScoreScreen.wnd:ButtonOk
click SkirmishGameOptionsMenu.wnd:ButtonBack
mainMenu
wait 60
quit
